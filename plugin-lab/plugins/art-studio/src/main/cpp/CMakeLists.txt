cmake_minimum_required(VERSION 3.31)
project(artstudio_text LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(archive "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/harfbuzz/harfbuzz-src-11.0.1.tar.xz")
file(SHA256 "${archive}" actual_hash)
if(NOT actual_hash STREQUAL "fec8a07e51d99f4c511eedebd02ff86dbfa744c7a212c825d8d443b101a9a6c8")
  message(FATAL_ERROR "Pinned HarfBuzz source checksum mismatch")
endif()
set(vendor "${CMAKE_CURRENT_BINARY_DIR}/vendor")
file(MAKE_DIRECTORY "${vendor}")
file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${vendor}")
set(hb "${vendor}/harfbuzz-11.0.1/src")
add_library(artstudio_text_0262 SHARED art_text_jni.cpp "${hb}/harfbuzz.cc")
target_include_directories(artstudio_text_0262 PRIVATE "${hb}")
target_compile_definitions(artstudio_text_0262 PRIVATE HB_NO_GETENV HB_NO_SETLOCALE HB_NO_MMAP HB_NO_ATEXIT)
target_compile_options(artstudio_text_0262 PRIVATE -fvisibility=hidden)
target_link_options(artstudio_text_0262 PRIVATE "-Wl,-z,max-page-size=16384")
